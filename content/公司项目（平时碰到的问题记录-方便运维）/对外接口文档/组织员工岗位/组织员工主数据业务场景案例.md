4、转换树结构

```
private List<AbossBranchOrganizationSimpleTreeDTO> changeBranchOrganizationSimpleTree(List<AbossBranchOrganizationSimpleTreeDTO> abossBranchOrganizationSimpleTreeDTOS) {
    Map<String, AbossBranchOrganizationSimpleTreeDTO> map = abossBranchOrganizationSimpleTreeDTOS.stream().collect(Collectors.toMap(dto -> dto.getBranchOrgCode(), dto -> dto, (k1, k2) -> k1));
    List<AbossBranchOrganizationSimpleTreeDTO> targetList = new ArrayList<>();
    Iterator<AbossBranchOrganizationSimpleTreeDTO> iterator = abossBranchOrganizationSimpleTreeDTOS.iterator();
    while (iterator.hasNext()) {
        AbossBranchOrganizationSimpleTreeDTO it = iterator.next();
        if (map.containsKey(it.getBranchOrgParentCode())) {
            AbossBranchOrganizationSimpleTreeDTO innerDto = map.get(it.getBranchOrgParentCode());
            List<AbossBranchOrganizationSimpleTreeDTO> child = innerDto.getChildBranchOrgList();
            if (CollectionUtils.isEmpty(child)) {
                child = new ArrayList<>();
                child.add(it);
                innerDto.setChildBranchOrgList(child);
            } else {
                child.add(it);
            }
        } else {
            targetList.add(it);
        }
    }
    return targetList;
}

## case4:OrgInfoDTO转换AbossBranchTreeResDTO

```
/**
     * orgInfoDTOList 转换 abossBranchTreeResDTOList
     *
     * @param orgInfoDTOList
     * @return
     */
public static List<AbossBranchTreeResDTO> convertListByManage(List<OrgInfoDTO> orgInfoDTOList) {
    //转换map,赋值branchOrgParentName
    Map<String, OrgInfoDTO> orgInfoDTOMap = orgInfoDTOList.stream().collect(Collectors.toMap(dto -> dto.getBranchOrgCode(), dto -> dto, (k1, k2) -> k1));
    //OrgInfoDTO 转换 AbossBranchTreeResDTO
    return orgInfoDTOList.stream().map(orgInfoDTO -> convertByManage(orgInfoDTO, orgInfoDTOMap)).collect(Collectors.toList());
}
```

**🚩🚩****如果仅仅需要某一种类型的组织，可以按照类型去获取对应的组织DTO，一个组织只会有一种类型。**

```
    /**
     * OrgInfoDTO 转换 AbossBranchTreeResDTO
     *
     * @param orgInfoDTO
     * @param orgInfoDTOMap
     * @return
     */
    private static AbossBranchTreeResDTO convertByManage(OrgInfoDTO orgInfoDTO, Map<String, OrgInfoDTO> orgInfoDTOMap) {
        AbossBranchTreeResDTO abossBranchTreeResDTO = new AbossBranchTreeResDTO();
        abossBranchTreeResDTO.setBranchOrgCode(orgInfoDTO.getBranchOrgCode());
        //获取基本信息
        getOrgBasicInfoByManage(orgInfoDTO, orgInfoDTOMap, abossBranchTreeResDTO);
        //获取经营组织
        if (Objects.nonNull(orgInfoDTO.getOrgOperationDetailDTO())) {
            getOperationOrgInfo(orgInfoDTO, abossBranchTreeResDTO);
        }
        //获取管理组织
        if (Objects.nonNull(orgInfoDTO.getOrgManageDetailDTO())) {
            OrgManageDetailDTO orgManageDetailDTO = orgInfoDTO.getOrgManageDetailDTO();
            abossBranchTreeResDTO.setIsManageOpreation(orgManageDetailDTO.getIsBothManageOperation());
            abossBranchTreeResDTO.setIsClaimsDept(orgManageDetailDTO.getIsClaimsDept());
            abossBranchTreeResDTO.setOrgIsSetIssue(orgManageDetailDTO.getIsSetIssue());
            abossBranchTreeResDTO.setBranchOrgUnitType(orgManageDetailDTO.getBranchOrgUnitTypeManage());
        }
        //获取销售组织
        if (Objects.nonNull(orgInfoDTO.getOrgSaleDetaiDTO())) {
            OrgSaleDetaiDTO orgSaleDetaiDTO = orgInfoDTO.getOrgSaleDetaiDTO();
            abossBranchTreeResDTO.setBranchOrgManagementChannel(orgSaleDetaiDTO.getBranchOrgOperationChannel());
            abossBranchTreeResDTO.setBranchOrgUnitType(orgSaleDetaiDTO.getBranchOrgUnitTypeSale());
        }
        return abossBranchTreeResDTO;
    }
